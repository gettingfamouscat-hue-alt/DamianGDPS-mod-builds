cmake_minimum_required(VERSION 3.25)
project(DamianModPorts LANGUAGES CXX)
set_property(GLOBAL PROPERTY JOB_POOLS port_compile=4)
set(CMAKE_JOB_POOL_COMPILE port_compile)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64")
endif()
set(GEODE_PACKAGE_ZSTD OFF)
set(GEODE_DONT_INSTALL_MODS ON)
set(GEODE_MODS_BEING_BUILT
    "geode.node-ids:v1.24.0;alphalaneous.editortab_api:v3.0.0-beta.5;km7dev.server_api:v5.0.0-beta.1;sabe.persistenceapi:v1.3.0"
    CACHE INTERNAL "Local build dependencies")
add_subdirectory($ENV{GEODE_SDK} geode)
foreach(api IN ITEMS "ServerAPI|km7dev.server_api" "GDPSSwitcher|km7dev.gdps-switcher" "PersistenceAPI|sabe.persistenceapi" "EditorTabAPI|alphalaneous.editortab_api")
    string(REPLACE "|" ";" parts "${api}")
    list(GET parts 0 folder)
    list(GET parts 1 id)
    file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/${folder}/include/" DESTINATION "${CMAKE_CURRENT_SOURCE_DIR}/headers/${id}")
endforeach()
include_directories(SYSTEM "${CMAKE_CURRENT_SOURCE_DIR}/headers")
set(GEODE6_PROJECTS "ServerAPI;GDPSSwitcher;HalfBlockTab;GoldenCoin;RateButton;MapMod;PersistenceAPI;PlatformerSaves;GrandpaDemon" CACHE STRING "Projects to build")
foreach(mod IN LISTS GEODE6_PROJECTS)
    add_subdirectory(${mod})
endforeach()
if(TARGET PlatformerSaves AND TARGET PersistenceAPI)
    target_link_libraries(PlatformerSaves PersistenceAPI)
endif()
